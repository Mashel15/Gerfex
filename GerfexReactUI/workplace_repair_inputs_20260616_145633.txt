import json, os, shutil, runpy, io, contextlib
from pathlib import Path
from datetime import datetime

PY_ROOT = Path(__file__).resolve().parent
GERFEX_ROOT = PY_ROOT / "GerfexIntegratedV1"
APP_HOME = Path(os.environ.get(
    "GERFEX_APP_HOME",
    "/data/data/com.mashel15.gerfex/files/gerfex_runtime_data"
))

ADMIN_DIR = APP_HOME / "_admin"
SNAP_DIR = APP_HOME / "_snapshots"
KEY_FILE = ADMIN_DIR / "admin_key.txt"
AUDIT = ADMIN_DIR / "admin_audit.log"

for d in [ADMIN_DIR, SNAP_DIR]:
    d.mkdir(parents=True, exist_ok=True)

def now():
    return datetime.utcnow().strftime("%Y%m%d_%H%M%S")

def trim_file_lines(path, keep=500):
    if not path.exists():
        return
    lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
    if len(lines) > keep:
        path.write_text("\n".join(lines[-keep:]) + "\n", encoding="utf-8")

def audit(action, path="", ok=True, note=""):
    with AUDIT.open("a", encoding="utf-8") as f:
        f.write(json.dumps({
            "time": datetime.utcnow().isoformat() + "Z",
            "action": action,
            "path": path,
            "ok": ok,
            "note": note
        }, ensure_ascii=False) + "\n")
    trim_file_lines(AUDIT, 500)

def check_key(req):
    key = req.get("api_key", "")
    if not KEY_FILE.exists():
        if req.get("action") == "init_key" and key:
            KEY_FILE.write_text(key, encoding="utf-8")
            audit("init_key", ok=True)
            return True, None
        return False, {"ok": False, "error": "admin_key_not_initialized"}
    stored = KEY_FILE.read_text(encoding="utf-8").strip()
    if key != stored:
        audit(req.get("action", "unknown"), ok=False, note="bad_api_key")
        return False, {"ok": False, "error": "bad_api_key"}
    return True, None

def safe_path(rel):
    rel = (rel or "").strip().lstrip("/")
    base = GERFEX_ROOT
    p = (base / rel).resolve()
    if not str(p).startswith(str(base.resolve())):
        raise Exception("unsafe_path")
    return p

def make_snapshot(path):
    if not path.exists():
        return ""
    stamp = now()
    rel = path.relative_to(GERFEX_ROOT)
    target = SNAP_DIR / stamp / rel
    target.parent.mkdir(parents=True, exist_ok=True)
    if path.is_dir():
        shutil.copytree(path, target, dirs_exist_ok=True)
    else:
        shutil.copy2(path, target)
    return str(target)

def list_files(req):
    p = safe_path(req.get("path", ""))
    items = []
    for x in sorted(p.iterdir(), key=lambda v: (not v.is_dir(), v.name.lower())):
        items.append({
            "name": x.name,
            "path": str(x.relative_to(GERFEX_ROOT)),
            "type": "dir" if x.is_dir() else "file",
            "size": x.stat().st_size if x.is_file() else 0
        })
    return {"ok": True, "root": str(GERFEX_ROOT), "items": items}

def read_file(req):
    p = safe_path(req.get("path", ""))
    return {"ok": True, "path": str(p.relative_to(GERFEX_ROOT)), "content": p.read_text(encoding="utf-8", errors="replace")}

def write_file(req):
    p = safe_path(req.get("path", ""))
    snap = make_snapshot(p)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(req.get("content", ""), encoding="utf-8")
    return {"ok": True, "path": str(p.relative_to(GERFEX_ROOT)), "snapshot": snap}

def delete_file(req):
    p = safe_path(req.get("path", ""))
    if not p.exists():
        return {"ok": False, "error": "file_not_found", "path": str(p)}
    snap = make_snapshot(p)
    if p.is_dir():
        shutil.rmtree(p)
    else:
        p.unlink()
    return {"ok": True, "deleted": str(p.relative_to(GERFEX_ROOT)), "snapshot": snap}


def create_snapshot(req):
    p = safe_path(req.get("path", ""))
    if not p.exists():
        return {"ok": False, "error": "file_not_found", "path": str(p)}
    snap = make_snapshot(p)
    return {"ok": True, "snapshot": snap}

def list_snapshots(req):
    items = []
    if SNAP_DIR.exists():
        for x in sorted(SNAP_DIR.rglob("*"), reverse=True):
            if x.is_file():
                items.append({"name": x.name, "path": str(x), "type": "file"})
            elif x.is_dir() and x != SNAP_DIR:
                items.append({"name": x.name, "path": str(x), "type": "dir"})
    return {"ok": True, "items": items[:50]}

def restore_snapshot(req):
    src = Path(req.get("snapshot", "")).resolve()
    if not str(src).startswith(str(SNAP_DIR.resolve())):
        raise Exception("unsafe_snapshot")
    rel = src.relative_to(SNAP_DIR)
    parts = rel.parts[1:]
    dst = (GERFEX_ROOT / Path(*parts)).resolve()
    make_snapshot(dst)
    if dst.exists():
        if dst.is_dir(): shutil.rmtree(dst)
        else: dst.unlink()
    dst.parent.mkdir(parents=True, exist_ok=True)
    if src.is_dir():
        shutil.copytree(src, dst)
    else:
        shutil.copy2(src, dst)
    return {"ok": True, "restored_to": str(dst)}

def add_lesson(req):
    p = safe_path("learning/pending_lessons.json")
    p.parent.mkdir(parents=True, exist_ok=True)
    data = []
    if p.exists():
        try: data = json.loads(p.read_text(encoding="utf-8"))
        except Exception: data = []
    snap = make_snapshot(p)
    lesson = req.get("lesson", {})
    lesson["time"] = datetime.utcnow().isoformat() + "Z"
    data.append(lesson)
    p.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
    return {"ok": True, "snapshot": snap, "count": len(data)}

def run_test(req):
    p = safe_path(req.get("path", ""))
    buf_out, buf_err = io.StringIO(), io.StringIO()
    try:
        with contextlib.redirect_stdout(buf_out), contextlib.redirect_stderr(buf_err):
            runpy.run_path(str(p), run_name="__main__")
        return {"ok": True, "stdout": buf_out.getvalue(), "stderr": buf_err.getvalue()}
    except Exception as e:
        return {"ok": False, "stdout": buf_out.getvalue(), "stderr": buf_err.getvalue(), "error": str(e)}

def handle(payload):
    try:
        req = json.loads(payload) if isinstance(payload, str) else payload
        allowed, err = check_key(req)
        if not allowed:
            return json.dumps(err, ensure_ascii=False)

        action = req.get("action")
        funcs = {
            "init_key": lambda r: {"ok": True, "message": "key_ready"},
            "list_files": list_files,
            "read_file": read_file,
            "write_file": write_file,
            "delete_file": delete_file,
            "add_lesson": add_lesson,
            "run_test": run_test,
            "read_trace": read_trace,
            "clear_trace": clear_trace,
            "create_snapshot": create_snapshot,
            "list_snapshots": list_snapshots,
            "restore_snapshot": restore_snapshot,
        }
        if action not in funcs:
            raise Exception("unknown_action: " + str(action))

        res = funcs[action](req)
        audit(action, req.get("path", ""), res.get("ok", True))
        return json.dumps(res, ensure_ascii=False)

    except Exception as e:
        audit("exception", ok=False, note=str(e))
        return json.dumps({"ok": False, "error": str(e)}, ensure_ascii=False)
